

@router.post("/{job_id}/application-route")
async def update_application_route(job_id: str, evidence: dict, db: Session = Depends(get_db)):
    """Enrich the same source posting using an authenticated browser's Apply control."""
    from backend.scraper._shared.linkedin_routing import linkedin_job_id, resolve_linkedin_route
    from backend.scraper.sources.jobspy import _backfill_route_fields
    job = db.query(Job).filter(Job.id == job_id).first()
    if not job:
        raise HTTPException(status_code=404, detail="Job not found")
    source_url = job.source_url or job.url
    expected = linkedin_job_id(source_url)
    if not expected or str(evidence.get("linkedin_job_id", "")) != expected:
        raise HTTPException(status_code=409, detail="Source posting identity mismatch")
    kind = evidence.get("application_type")
    if kind not in ("linkedin_easy_apply", "external") or evidence.get("evidence_source") != "jobbot_owned_browser_apply_control":
        raise HTTPException(status_code=422, detail="Observed Apply control evidence required")
    route = resolve_linkedin_route(source_url, evidence.get("application_url"),
        easy_apply_observed=kind == "linkedin_easy_apply", external_expected=kind == "external",
        evidence_source="jobbot_owned_browser_apply_control")
    if route["resolution_status"] not in ("resolved", "not_required"):
        raise HTTPException(status_code=422, detail="Public application destination required")
    route["application_type_evidence"]["source"] = "jobbot_owned_browser_apply_control"
    route["application_type_evidence"]["linkedin_job_id"] = expected
    _backfill_route_fields(job, route)
    db.commit()
    return {"id": str(job.id), "application_type": job.application_type,
            "resolution_status": job.resolution_status, "application_url": job.application_url}
